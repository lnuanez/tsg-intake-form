/**
 * TSG — Attach Resume to Gmail Draft (v2)
 * =========================================
 * Changes from v1:
 *   - Duplicate check: skips attachment if PDF already present
 *   - Accepts ?candidate=Name param from Claude for precise matching
 *   - doGet() handler so Claude can trigger via web app URL
 *   - installTrigger() fallback still available
 */

var CONFIG = {
  RESUME_FOLDER_ID: "1h3xazJKaPbYfjkp_7pB_Cq4HPoMafOK6",
  DRAFT_LOOKBACK_MINUTES: 30,
};

// ─── WEB APP ENTRY POINT ─────────────────────────────────────────────────────
// Called by Claude via: GET {webAppUrl}?candidate=Fausto+Zambrano

function doGet(e) {
  var candidateName = e && e.parameter && e.parameter.candidate
    ? decodeURIComponent(e.parameter.candidate)
    : null;

  var result = attachResumeToDraft(candidateName);

  return ContentService
    .createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

// ─── MAIN FUNCTION ───────────────────────────────────────────────────────────

function attachResumeToDraft(candidateNameOverride) {
  Logger.log("=== TSG Resume Attachment Script v2 ===");

  // Step 1: Find the most recent unprocessed submission draft
  var draft = findLatestSubmissionDraft(candidateNameOverride);
  if (!draft) {
    var msg = "❌ No recent submission draft found.";
    Logger.log(msg);
    return { success: false, error: msg };
  }

  var subject = draft.getMessage().getSubject();
  Logger.log("✓ Found draft: " + subject);

  // Step 2: Check if PDF already attached — abort if so
  if (draftHasPdfAttachment(draft)) {
    var msg = "⚠️ PDF already attached to this draft. Skipping.";
    Logger.log(msg);
    return { success: true, skipped: true, reason: msg };
  }

  // Step 3: Get candidate name — prefer override, fall back to subject parse
  var candidateName = candidateNameOverride || extractCandidateName(subject);
  if (!candidateName) {
    var msg = "❌ Could not extract candidate name from: " + subject;
    Logger.log(msg);
    return { success: false, error: msg };
  }
  Logger.log("✓ Candidate: " + candidateName);

  // Step 4: Find resume PDF in Drive
  var resumeFile = findResumeInDrive(candidateName);
  if (!resumeFile) {
    var msg = "❌ No resume PDF found for: " + candidateName;
    Logger.log(msg);
    return { success: false, error: msg };
  }
  Logger.log("✓ Resume file: " + resumeFile.getName());

  // Step 5: Attach PDF to draft
  var success = attachFileToDraft(draft, resumeFile);
  if (success) {
    Logger.log("✅ Resume attached! Draft ready to review and send.");
    return { success: true, candidate: candidateName, file: resumeFile.getName() };
  } else {
    return { success: false, error: "Attachment failed — see execution log." };
  }
}

// ─── DUPLICATE CHECK ─────────────────────────────────────────────────────────

function draftHasPdfAttachment(draft) {
  try {
    var attachments = draft.getMessage().getAttachments();
    for (var i = 0; i < attachments.length; i++) {
      if (attachments[i].getContentType() === "application/pdf") {
        return true;
      }
    }
  } catch (e) {
    Logger.log("Warning: could not check attachments: " + e.message);
  }
  return false;
}

// ─── FIND DRAFT ──────────────────────────────────────────────────────────────

function findLatestSubmissionDraft(candidateNameOverride) {
  var drafts = GmailApp.getDrafts();
  var cutoff = new Date();
  cutoff.setMinutes(cutoff.getMinutes() - CONFIG.DRAFT_LOOKBACK_MINUTES);

  var latestDraft = null;
  var latestDate = null;

  for (var i = 0; i < drafts.length; i++) {
    var msg = drafts[i].getMessage();
    var subject = msg.getSubject();
    var date = msg.getDate();

    // Must have " - " separator (e.g. "Fausto Zambrano - Full Stack Dev")
    if (subject.indexOf(" - ") === -1) continue;

    // If we have a name override, filter to only matching drafts
    if (candidateNameOverride) {
      var nameLower = candidateNameOverride.toLowerCase();
      var subjectLower = subject.toLowerCase();
      var nameParts = nameLower.split(" ");
      var matches = nameParts.some(function(p) {
        return p.length > 2 && subjectLower.indexOf(p) !== -1;
      });
      if (!matches) continue;
    } else {
      // No override — only look within the lookback window
      if (date < cutoff) continue;
    }

    // Skip if already has PDF attached
    if (draftHasPdfAttachment(drafts[i])) continue;

    if (!latestDate || date > latestDate) {
      latestDraft = drafts[i];
      latestDate = date;
    }
  }

  return latestDraft;
}

// ─── EXTRACT CANDIDATE NAME ──────────────────────────────────────────────────

function extractCandidateName(subject) {
  var parts = subject.split(" - ");
  return parts.length >= 2 ? parts[0].trim() : null;
}

// ─── FIND RESUME IN DRIVE ────────────────────────────────────────────────────

function findResumeInDrive(candidateName) {
  var folder;
  try {
    folder = DriveApp.getFolderById(CONFIG.RESUME_FOLDER_ID);
  } catch (e) {
    Logger.log("❌ Cannot access Drive folder: " + e.message);
    return null;
  }

  var files = folder.getFilesByType(MimeType.PDF);
  var nameParts = candidateName.toLowerCase().split(" ");
  var bestMatch = null;
  var bestScore = 0;

  while (files.hasNext()) {
    var file = files.next();
    var fileName = file.getName().toLowerCase();
    var score = 0;
    for (var i = 0; i < nameParts.length; i++) {
      if (nameParts[i].length > 2 && fileName.indexOf(nameParts[i]) !== -1) score++;
    }
    if (score > bestScore) { bestScore = score; bestMatch = file; }
  }

  return bestScore > 0 ? bestMatch : null;
}

// ─── ATTACH FILE TO DRAFT ────────────────────────────────────────────────────

function attachFileToDraft(draft, file) {
  try {
    var msg = draft.getMessage();
    var to = msg.getTo();
    var cc = msg.getCc();
    var subject = msg.getSubject();
    var body = msg.getBody();
    var plainBody = msg.getPlainBody();

    // Double-check no PDF before deleting old draft
    var existing = msg.getAttachments();
    for (var i = 0; i < existing.length; i++) {
      if (existing[i].getContentType() === "application/pdf") {
        Logger.log("⚠️ PDF already attached — aborting to prevent duplicate.");
        return false;
      }
    }

    var blob = file.getBlob();
    Logger.log("  Attaching: " + file.getName() + " (" + Math.round(blob.getBytes().length / 1024) + " KB)");

    draft.deleteDraft();

    GmailApp.createDraft(to, subject, plainBody, {
      attachments: [blob],
      cc: cc,
      htmlBody: body
    });

    Logger.log("  ✅ New draft created with attachment.");
    return true;
  } catch (e) {
    Logger.log("❌ Error: " + e.message);
    return false;
  }
}

// ─── TRIGGER SETUP ───────────────────────────────────────────────────────────

function installTrigger() {
  removeTrigger();
  ScriptApp.newTrigger("attachResumeToDraft")
    .timeBased()
    .everyMinutes(5)
    .create();
  Logger.log("✅ Trigger installed — runs every 5 minutes.");
}

function removeTrigger() {
  var triggers = ScriptApp.getProjectTriggers();
  for (var i = 0; i < triggers.length; i++) ScriptApp.deleteTrigger(triggers[i]);
  Logger.log("✅ All triggers removed.");
}

// ─── DIAGNOSTIC ──────────────────────────────────────────────────────────────

function listRecentDrafts() {
  var drafts = GmailApp.getDrafts();
  Logger.log("Recent drafts (" + drafts.length + " total):");
  for (var i = 0; i < Math.min(drafts.length, 10); i++) {
    var msg = drafts[i].getMessage();
    var hasPdf = draftHasPdfAttachment(drafts[i]) ? " [HAS PDF]" : "";
    Logger.log("  [" + i + "] " + msg.getDate() + " | " + msg.getSubject() + hasPdf);
  }
}
